"""Configuration helpers for IO store construction."""

import warnings
from dataclasses import dataclass, field, fields
from enum import Enum
from pathlib import Path
from typing import Any

_SIZE_SUFFIXES = {
    "tb": 1024**4,
    "gb": 1024**3,
    "mb": 1024**2,
    "kb": 1024**1,
    "b": 1024**0,
}
_PARQUET_RG_CACHE_MIN_BYTES = 4 * 1024**3
_OPTION_DEFAULT = "zephon_default"


class _NotSupplied(Enum):
    TOKEN = 0


_NOT_SUPPLIED = _NotSupplied.TOKEN


def _option(default: object, *, preserve_none: bool = False) -> Any:
    return field(
        default=_NOT_SUPPLIED,
        metadata={_OPTION_DEFAULT: default, "preserve_none": preserve_none},
    )


def _normalize_options(options: Any) -> None:
    provided = set()
    for option in fields(options):
        if _OPTION_DEFAULT not in option.metadata:
            continue
        value = getattr(options, option.name)
        if value is not _NOT_SUPPLIED:
            provided.add(option.name)
        if value is _NOT_SUPPLIED or (
            value is None and not option.metadata["preserve_none"]
        ):
            setattr(options, option.name, option.metadata[_OPTION_DEFAULT])
    options._provided_fields = frozenset(provided)


def _merged_options(base: Any, override: Any) -> dict[str, Any]:
    merged = {}
    for option in fields(base):
        if _OPTION_DEFAULT not in option.metadata:
            continue
        if option.name in override._provided_fields:
            merged[option.name] = getattr(override, option.name)
        elif option.name in base._provided_fields:
            merged[option.name] = getattr(base, option.name)
    return merged


def _default_options(option_type: Any) -> Any:
    values = {
        option.name: option.metadata[_OPTION_DEFAULT]
        for option in fields(option_type)
        if _OPTION_DEFAULT in option.metadata
    }
    return option_type(**values)


def _validate_non_negative_int(name: str, value: object) -> None:
    if isinstance(value, bool) or not isinstance(value, int) or value < 0:
        raise ValueError(f"{name} must be a non-negative integer")


def _validate_positive_int(name: str, value: object) -> None:
    if isinstance(value, bool) or not isinstance(value, int) or value < 1:
        raise ValueError(f"{name} must be a positive integer")


def parse_size_bytes(value: str | int | None) -> int | None:
    """Parse a human-readable size like ``"750gb"`` into a byte count.

    Accepts ``None`` (returns ``None``), an int (returned as-is), or a string
    with an optional suffix in {tb, gb, mb, kb, b}. Empty or whitespace-only
    strings return ``None``. Raises ``ValueError`` on unparseable input.
    """
    if value is None:
        return None
    if isinstance(value, int):
        return value
    if not isinstance(value, str):
        raise TypeError(f"Cannot parse size from {value!r}")
    s = value.strip().lower()
    if not s:
        return None
    for suffix, multiplier in _SIZE_SUFFIXES.items():
        if s.endswith(suffix):
            return int(float(s[: -len(suffix)]) * multiplier)
    return int(s)

@dataclass
class CacheOptions:
    """On-disk shard-cache configuration.

    Supplying a nonempty ``root`` enables caching when ``enabled`` is omitted.
    An explicit ``enabled=False`` takes precedence, including across option
    merges. Without a supplied root or enablement, caching stays disabled.
    Configure ``limit_bytes`` to bound disk usage; its default is unlimited.
    """

    enabled: bool = _option(False)
    root: str | Path = _option(Path("~/.cache/zephon").expanduser())
    # Bound on the on-disk shard cache only, not in-memory cache.
    limit_bytes: int | None = _option(None)
    # Deprecated alias migrated by StoreOptions to parquet.rg_cache.limit_bytes.
    rg_cache_bytes: int | None = _option(None)
    keep_zip: bool = _option(False)
    validate_hash: str | None = _option(None)
    download_retry: int = _option(12)
    download_timeout: float = _option(180.0)
    open_retry_attempts: int = _option(5)
    open_retry_initial_backoff: float = _option(0.1)
    open_retry_max_backoff: float = _option(2.0)
    min_slack_bytes: int = _option(512 * 1024)  # 512 KiB minimum slack
    max_slack_bytes: int = _option(64 * 1024 * 1024)  # 64 MiB maximum slack
    # Keep shards across runs; reused only if the next run reads the same shards.
    persist_state: bool = _option(False)
    _provided_fields: frozenset[str] = field(
        init=False,
        repr=False,
        compare=False,
    )
    _root_was_reset: bool = field(init=False, repr=False, compare=False)

    def __post_init__(self) -> None:
        # Raw dataclass inputs can include the None reset before normalization.
        raw_root: Any = self.root
        self._root_was_reset = raw_root is None
        _normalize_options(self)
        if (
            "root" in self._provided_fields
            and "enabled" not in self._provided_fields
            and not self._root_was_reset
            and self.root
        ):
            # Keep inferred enablement out of _provided_fields so a root-only
            # override cannot replace an explicit disable during a merge.
            self.enabled = True



[docs]
    @classmethod
    def from_any(cls, obj: Any) -> "CacheOptions":
        if obj is None or obj is False:
            return _default_options(cls)
        if isinstance(obj, CacheOptions):
            return obj
        if isinstance(obj, dict):
            data = dict(obj)
            if "limit_bytes" in data and isinstance(data["limit_bytes"], str):
                data["limit_bytes"] = parse_size_bytes(data["limit_bytes"])
            if "rg_cache_bytes" in data and isinstance(data["rg_cache_bytes"], str):
                data["rg_cache_bytes"] = parse_size_bytes(data["rg_cache_bytes"])
            return cls(**data)
        raise TypeError(f"Cannot interpret cache options from {obj!r}")





[docs]
    def merge(self, other: "CacheOptions") -> "CacheOptions":
        """Overlay fields supplied by ``other`` onto ``self``."""
        merged = _merged_options(self, other)
        root_source = other if "root" in other._provided_fields else self
        if root_source._root_was_reset:
            # Preserve a root reset rather than inferring enablement from the
            # default path installed by _normalize_options.
            merged["root"] = None
        return CacheOptions(**merged)

@dataclass
class ParquetRGCacheOptions:
    """Node-shared decoded Parquet row-group cache configuration.

    ``enabled=None`` enables the cache with the shard cache or an explicit
    ``root``. An omitted root uses the shard cache's reserved
    ``.parquet-rg-cache`` child. An omitted limit uses the deprecated shard
    option when present, otherwise the larger of 4 GiB and 10% of the shard
    cache limit. These defaults resolve when the store is built so option merges
    retain the distinction between omitted and explicit values. Explicit
    ``None`` resets a previously supplied field to its automatic default.
    """

    enabled: bool | None = _option(None)
    root: str | Path | None = _option(None)
    limit_bytes: int | None = _option(None)
    min_free_bytes: int = _option(1 * 1024**3)
    _provided_fields: frozenset[str] = field(
        init=False,
        repr=False,
        compare=False,
    )

    def __post_init__(self) -> None:
        _normalize_options(self)
        if (
            self.enabled is not False
            and self.limit_bytes is not None
            and self.limit_bytes <= 0
        ):
            raise ValueError("parquet.rg_cache.limit_bytes must be positive")
        if self.min_free_bytes < 0:
            raise ValueError("parquet.rg_cache.min_free_bytes cannot be negative")



[docs]
    @classmethod
    def from_any(cls, obj: Any) -> "ParquetRGCacheOptions":
        """Normalize a decoded row-group cache configuration."""
        if obj is None:
            return _default_options(cls)
        if obj is False:
            return cls(enabled=False)
        if obj is True:
            return cls(enabled=True)
        if isinstance(obj, ParquetRGCacheOptions):
            return obj
        if isinstance(obj, dict):
            data = dict(obj)
            for field_name in ("limit_bytes", "min_free_bytes"):
                value = data.get(field_name)
                if isinstance(value, str):
                    parsed = parse_size_bytes(value)
                    if parsed is None:
                        raise ValueError(
                            f"parquet.rg_cache.{field_name} must not be empty"
                        )
                    data[field_name] = parsed
            return cls(**data)
        raise TypeError(f"Cannot interpret Parquet RG cache options from {obj!r}")





[docs]
    def merge(self, other: "ParquetRGCacheOptions") -> "ParquetRGCacheOptions":
        """Overlay fields supplied by ``other`` onto ``self``."""
        return ParquetRGCacheOptions(**_merged_options(self, other))

class ArrowOptions:
    """Process-wide Arrow CPU and I/O pool sizes, shared by all Arrow readers.

    Counts default to one. Explicit ``None`` leaves that pool unchanged.
    Configuring another store in the same process affects existing readers too.
    """

    cpu_threads: int | None = _option(1, preserve_none=True)
    io_threads: int | None = _option(1, preserve_none=True)
    _provided_fields: frozenset[str] = field(init=False, repr=False, compare=False)

    def __post_init__(self) -> None:
        _normalize_options(self)
        for name in ("cpu_threads", "io_threads"):
            value = getattr(self, name)
            if value is not None:
                _validate_non_negative_int(f"arrow.{name}", value)
                if value == 0:
                    raise ValueError(f"arrow.{name} must be positive or None")



[docs]
    @classmethod
    def from_any(cls, obj: Any) -> "ArrowOptions":
        """Normalize Arrow pool options without importing PyArrow."""
        if obj is None:
            return _default_options(cls)
        if isinstance(obj, cls):
            return obj
        if isinstance(obj, dict):
            return cls(**obj)
        raise TypeError(f"Cannot interpret Arrow options from {obj!r}")

        """Overlay supplied fields, preserving explicit ``None``."""
        return ArrowOptions(**_merged_options(self, other))

@dataclass
class ParquetOptions:
    """Parquet read behavior and the node-shared decoded row-group cache.

    ``use_threads`` enables parallel column decoding through Arrow's CPU pool.
    False keeps each read on its calling thread; separate fetches can still
    decode concurrently. Pool sizes are configured separately in ArrowOptions.
    """

    use_threads: bool = _option(False)
    rg_cache: ParquetRGCacheOptions = field(default_factory=ParquetRGCacheOptions)
    _provided_fields: frozenset[str] = field(init=False, repr=False, compare=False)

    def __post_init__(self) -> None:
        _normalize_options(self)
        if type(self.use_threads) is not bool:
            raise ValueError("parquet.use_threads must be a boolean")



[docs]
    @classmethod
    def from_any(cls, obj: Any) -> "ParquetOptions":
        """Normalize Parquet read and cache options."""
        if obj is None:
            return cls(use_threads=False, rg_cache=ParquetRGCacheOptions.from_any(None))
        if isinstance(obj, cls):
            return obj
        if isinstance(obj, dict):
            data = dict(obj)
            if "rg_cache" in data:
                data["rg_cache"] = ParquetRGCacheOptions.from_any(data["rg_cache"])
            return cls(**data)
        raise TypeError(f"Cannot interpret Parquet options from {obj!r}")

@dataclass
class VortexOptions:
    """Vortex read-cache and process-wide worker-pool configuration.

    ``memory_map`` defaults to False, using file IO and the segment cache.
    True maps local shards and reads segments as slices of the map, using the
    OS page cache instead. Decoding and output conversion can still allocate
    and copy. Mapped shards must not change while in use.
    ``safe_mmap`` copies array views from mapped reads into owned memory so
    retained samples cannot keep evicted shard files on disk. Process and Ray
    stages skip this extra copy because sending outputs already copies mapped
    arrays into independent storage. False keeps array views on thread and
    inline runners.
    ``segment_cache_bytes`` sets the encoded-segment cache's target size per
    fetch replica when ``memory_map`` is False. Zero disables it. The target is
    not a strict memory bound.
    ``worker_threads`` defaults to one background worker; calling fetch threads
    can also execute Vortex work. Zero disables background workers, and explicit
    None leaves the pool unchanged. An explicit count overrides VORTEX_MAX_THREADS;
    an omitted count respects that environment variable. Stores in one process
    share the pool, so the last applied count wins.
    """

    segment_cache_bytes: int = _option(64 * 1024**2)
    worker_threads: int | None = _option(1, preserve_none=True)
    # Mmap can help with large payloads such as images, but can slow down
    # token and text reads. Measure your workload before enabling it.
    memory_map: bool = _option(False)
    safe_mmap: bool = _option(True)
    _provided_fields: frozenset[str] = field(
        init=False,
        repr=False,
        compare=False,
    )

    def __post_init__(self) -> None:
        _normalize_options(self)
        if type(self.memory_map) is not bool:
            raise ValueError("vortex.memory_map must be a boolean")
        if type(self.safe_mmap) is not bool:
            raise ValueError("vortex.safe_mmap must be a boolean")

        if self.worker_threads is not None:
            _validate_non_negative_int("vortex.worker_threads", self.worker_threads)
        _validate_non_negative_int(
            "vortex.segment_cache_bytes", self.segment_cache_bytes
        )



[docs]
    @classmethod
    def from_any(cls, obj: Any) -> "VortexOptions":
        """Normalize Vortex options, accepting byte sizes such as ``'64mb'``."""
        if obj is None:
            return _default_options(cls)
        if isinstance(obj, VortexOptions):
            return obj
        if isinstance(obj, dict):
            data = dict(obj)
            if isinstance(data.get("segment_cache_bytes"), str):
                parsed = parse_size_bytes(data["segment_cache_bytes"])
                if parsed is None:
                    raise ValueError("vortex.segment_cache_bytes must not be empty")
                data["segment_cache_bytes"] = parsed
            return cls(**data)
        raise TypeError(f"Cannot interpret Vortex options from {obj!r}")

@dataclass
class LanceOptions:
    """Lance's process-wide worker pools.

    ``worker_threads`` (default 2) caps each of pylance's native pools, which
    otherwise start one worker per core; None leaves the existing pool
    configuration unchanged. pylance sizes them once per process, so the count
    applies only before its first use. An explicit count overrides
    TOKIO_WORKER_THREADS and LANCE_CPU_THREADS; an omitted one respects them.
    """

    worker_threads: int | None = _option(2, preserve_none=True)
    _provided_fields: frozenset[str] = field(
        init=False,
        repr=False,
        compare=False,
    )

    def __post_init__(self) -> None:
        _normalize_options(self)
        if self.worker_threads is not None:
            # pylance's runtime panics on a zero-sized pool.
            _validate_positive_int("lance.worker_threads", self.worker_threads)

class StoreOptions:
    """Top-level IO store options passed to FetchOp.

    ``metadata_cache_entries`` bounds retained footers per format and fetch
    replica, independently of the on-disk shard cache; zero disables retention.
    Parquet footer sizes grow with the number of columns and row groups.
    """

    cache: CacheOptions = field(default_factory=CacheOptions)
    arrow: ArrowOptions = field(default_factory=ArrowOptions)
    parquet: ParquetOptions = field(default_factory=ParquetOptions)
    vortex: VortexOptions = field(default_factory=VortexOptions)
    lance: LanceOptions = field(default_factory=LanceOptions)
    metadata_cache_entries: int = _option(256)
    _provided_fields: frozenset[str] = field(init=False, repr=False, compare=False)

    def __post_init__(self) -> None:
        _normalize_options(self)
        _validate_non_negative_int(
            "metadata_cache_entries", self.metadata_cache_entries
        )
        legacy_limit = self.cache.rg_cache_bytes
        if legacy_limit is not None:
            warnings.warn(
                "cache.rg_cache_bytes is deprecated; use "
                + "parquet.rg_cache.limit_bytes",
                DeprecationWarning,
                stacklevel=2,
            )

    def resolved_parquet_rg_cache(self) -> ParquetRGCacheOptions:
        """Resolve decoded-cache defaults against the shard-cache options."""
        configured = self.parquet.rg_cache
        if (
            configured.enabled is True
            and not self.cache.enabled
            and configured.root is None
        ):
            raise ValueError(
                "parquet.rg_cache.enabled=True requires parquet.rg_cache.root "
                + "when cache.enabled=False"
            )
        legacy_limit = self.cache.rg_cache_bytes
        limit = configured.limit_bytes
        enabled = (
            configured.enabled
            if configured.enabled is not None
            else self.cache.enabled or configured.root is not None
        )
        if limit is None:
            if legacy_limit is not None:
                enabled = enabled and legacy_limit > 0
                limit = (
                    legacy_limit if legacy_limit > 0 else _PARQUET_RG_CACHE_MIN_BYTES
                )
            else:
                shard_limit = self.cache.limit_bytes if self.cache.enabled else None
                limit = max(
                    _PARQUET_RG_CACHE_MIN_BYTES,
                    shard_limit // 10 if shard_limit is not None else 0,
                )
        return ParquetRGCacheOptions(
            enabled=enabled,
            root=configured.root,
            limit_bytes=limit,
            min_free_bytes=configured.min_free_bytes,
        )
